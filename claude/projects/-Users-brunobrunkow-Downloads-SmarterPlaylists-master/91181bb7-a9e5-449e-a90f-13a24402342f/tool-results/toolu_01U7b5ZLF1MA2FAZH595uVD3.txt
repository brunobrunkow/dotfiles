    70→                return track
    71→            elif not self.fail_fast:
    72→                self.source_list.remove(src)

   104→class TrackFilter(object):
   105→    '''
   106→        produces tracks on the true source that are not on the false source
   107→    '''
   108→    def __init__(self, true_source, false_source, invert=False, by_name=False):
   109→        prep = ' that are also in ' if invert else ' that are not in '
   110→        self.name = 'tracks in ' + true_source.name  + prep + \
   111→            false_source.name
   112→        self.true_source = true_source
   113→        self.false_source = false_source
   114→        self.bad_tracks = set()
   115→        self.invert = invert
   116→        self.debug = False
   117→        self.by_name = by_name
   118→
   119→    def next_track(self):
   120→        while True:
   121→            bad_track = self.false_source.next_track()
   122→            self.bad_tracks.add(bad_track)
   123→            if bad_track:
   124→                if self.by_name:
   125→                    bad_track_name = pbl.tlib.get_tn(bad_track).lower()
   126→                    self.bad_tracks.add(bad_track_name)
   127→            else:
   128→                break
   129→
   130→        while True:
   131→            track = self.true_source.next_track()
   132→            if track:
   133→                if self.by_name:
   134→                    track_name = pbl.tlib.get_tn(track).lower()
   135→                else:
   136→                    track_name = track
   137→
   138→                if self.invert and ((track in self.bad_tracks) or (track_name in self.bad_tracks)):
   139→                    return track
   140→                elif (not self.invert) and ((track not in self.bad_tracks) and (track_name not in self.bad_tracks)):
   141→                    return track
   142→                else:
   143→                    if self.debug:
   144→                        print('filtered out', pbl.tlib.get_tn(track))
   145→            else:
   146→                break
   147→        return None

   187→class TextFilter(object):
   188→    '''
   189→        produces tracks from the stream based on track title match
   190→    '''
   191→    def __init__(self, source, text, ignore_case=False, invert=False):
   192→        prep = ' that do not match ' if invert else ' that match '
   193→        self.name = 'tracks in ' + source.name  + prep + '"' + text + '"'
   194→        self.source = source
   195→        self.invert = invert
   196→        flags = re.UNICODE
   197→        if ignore_case:
   198→            flags |= re.IGNORECASE
   199→        self.regex = re.compile(text, flags)
   200→
   201→    def next_track(self):
   202→        while True:
   203→            track = self.source.next_track()
   204→            if track:
   205→                tinfo = pbl.tlib.get_track(track)
   206→                title = tinfo['title']
   207→                does_match = self.regex.search(title) != None
   208→                if does_match == self.invert:
   209→                    continue
   210→                else:
   211→                    return track
   212→            else:
   213→                break
   214→        return None

   300→class AllButTheFirst(object):
   301→    '''
   302→        Returns all but the first tracks from a stream
   303→
   304→        :param source: the source of tracks
   305→        :param sample_size: the number of tracks to skip
   306→    '''
   307→    def __init__(self, source, sample_size=10):
   308→        self.name = 'all but the first ' + str(sample_size) + ' of ' + source.name
   309→        self.source = source
   310→        self.sample_size = sample_size
   311→        self.buffer = []
   312→        self.filling = True
   313→
   314→    def next_track(self):
   315→        while self.filling:
   316→            track = self.source.next_track()
   317→            if track:
   318→                self.buffer.append(track)
   319→            else:
   320→                self.filling = False
   321→                self.buffer = self.buffer[self.sample_size:]
   322→
   323→        if len(self.buffer) > 0:
   324→            return self.buffer.pop(0)
   325→        else:
   326→            return None

   386→        self.source = source
   387→        self.name = source.name + ' saved to ' + playlist_name
   388→        self.playlist_name = playlist_name

   399→        track = self.source.next_track()
   400→        if track and len(self.buffer) < self.max_size:
   401→            self.buffer.append(track)
   402→        elif not self.saved:
   403→            self._save_playlist()
   404→        return track

   414→        if not user:
   415→            raise pbl.PBLException(self, "no user")

   417→        # TODO - this can be slow if the user has lots of playlists
   418→        # should cache?

   420→            uri = self.playlist_uri
   421→            pid = get_pid_from_playlist_uri(uri)
   422→
   423→            if not user:
   424→                raise pbl.PBLException(self, "bad uri")
   425→
   426→            if not pid:
   427→                raise pbl.PBLException(self, "bad uri")

   429→            uri = find_playlist_by_name(sp, user, self.playlist_name)

   431→        if uri:

   435→            response = sp.user_playlist_create(user, self.playlist_name)

   438→        pid = get_pid_from_playlist_uri(uri)

   580→class MySavedTracks(object):
   581→    ''' A PBL Source that generates a list of the saved tracks
   582→        by the current suer
   583→    '''
   584→
   585→    def __init__(self):
   586→        self.name = 'My Saved Tracks'
   587→        self.buffer = None
   588→
   589→    def next_track(self):
   590→        if self.buffer == None:
   591→            self.buffer = []
   592→            try:
   593→                sp = get_spotify()
   594→                limit = 50
   595→                offset = 0
   596→                total = 50
   597→
   598→                while offset < total:
   599→                    results = sp.current_user_saved_tracks(limit = limit, offset = offset)
   600→                    for item in results['items']:
   601→                        track = item['track']
   602→                        if track and 'id' in track:

   604→                            spotify_plugs._add_track(self.name, track)
   605→                        else:
   606→                            raise pbl.engine.PBLException(self, 'bad track')

   969→class MixIn(object):
   970→    '''
   971→        A PBL Filter that mixes two input streams based upon a small
   972→        set of rules
   973→    '''
   974→
   975→    def __init__(self, true_source, false_source,
   976→        ntracks=1, nskips=1, initial_skip = 1, fail_fast = True):
   977→        '''
   978→            params:
   979→                * true_source
   980→                * false_source
   981→        '''
   982→        self.name = 'mixing ' + true_source.name  + ' with ' + \
   983→            false_source.name
   984→        self.true_source = true_source
   985→        self.false_source = false_source
   986→        self.initial_offset = initial_skip
   987→        self.ntracks = ntracks # false tracks in a row
   988→        self.nskips = nskips   # true tracks in a row
   989→        self.fail_fast = fail_fast
   990→        self.which = 0
   991→
   992→    def next_track(self):
   993→
   994→        if self.which < self.initial_offset:
   995→            true_count = self.initial_offset
   996→        else:
   997→            true_count = self.nskips
   998→
   999→        false_count = self.ntracks
  1000→        total_tracks_in_cycle = true_count + false_count
  1001→        which_in_cycle = self.which % total_tracks_in_cycle
  1002→        use_first = which_in_cycle < true_count
  1003→
  1004→        if use_first:
  1005→            next_source = self.true_source
  1006→            other_source = self.false_source
  1007→        else:
  1008→            next_source = self.false_source
  1009→            other_source = self.true_source
  1010→
  1011→        track = next_source.next_track()
  1012→        if not track:
  1013→            if not self.fail_fast:
  1014→                track = other_source.next_track()
  1015→        self.which += 1
  1016→        return track

  1018→class SeparateArtists(object):
  1019→    ''' A PBL filter that reorders the input tracks to maximize
  1020→        the separation between artists
  1021→    '''
  1022→
  1023→    def __init__(self, source):
  1024→        self.name = 'SeparateArtists'
  1025→        self.source = source
  1026→        self.tracks = []
  1027→        self.filling = True
  1028→
  1029→    def score_list(self):
  1030→        min_delta_allowed = 2
  1031→        score = 0
  1032→        indexes = set()
  1033→
  1034→        for i in range(len(self.tracks) - 1):
  1035→            aname = self.tracks[i]['artist']
  1036→            for j in range(i + 1, len(self.tracks)):
  1037→
  1038→                if j - i >= min_delta_allowed:
  1039→                    break
  1040→
  1041→                bname = self.tracks[j]['artist']
  1042→                if aname == bname:
  1043→                    delta = j - i
  1044→                    indexes.add(i)
  1045→                    indexes.add(j)
  1046→                    score += min_delta_allowed - delta
  1047→
  1048→        lindex = list(indexes)
  1049→        lindex.sort()
  1050→        return score, lindex
  1051→
  1052→    def random_index(self):
  1053→        return random.randint(0, len(self.tracks) - 1)
  1054→
  1055→    def swap(self, a, b):
  1056→        tmp = self.tracks[a]
  1057→        self.tracks[a] = self.tracks[b]
  1058→        self.tracks[b] = tmp
  1059→
  1060→    def separate_artists(self):
  1061→        max_tries = 1000
  1062→        cur_score = 0
  1063→        max_no_swaps = 100
  1064→
  1065→        cur_score, indexes = self.score_list()
  1066→        no_swap = 0
  1067→
  1068→        for i in range(max_tries):
  1069→
  1070→            if cur_score == 0 or len(indexes) == 0:
  1071→                break
  1072→
  1073→            swap_1 = random.choice(indexes)
  1074→            swap_2 = self.random_index()
  1075→            self.swap(swap_1, swap_2)
  1076→            new_score, new_indexes = self.score_list()
  1077→
  1078→            if new_score >= cur_score:
  1079→                self.swap(swap_2, swap_1)
  1080→                no_swap += 1
  1081→                if no_swap > max_no_swaps:
  1082→                    break
  1083→            else:
  1084→                no_swap = 0
  1085→                cur_score = new_score
  1086→                indexes = new_indexes
  1087→
  1088→            if cur_score == 0:
  1089→                break
  1090→
  1091→        return cur_score, len(indexes)
  1092→
  1093→    def create_buffer(self):
  1094→        self.buffer = []
  1095→        for ti in self.tracks:
  1096→            self.buffer.append(ti['id'])
  1097→
  1098→    def next_track(self):
  1099→        while self.filling:
  1100→            track = self.source.next_track()
  1101→            if track:
  1102→                tinfo = pbl.tlib.get_track(track)
  1103→                self.tracks.append(tinfo)
  1104→            else:
  1105→                self.filling = False
  1106→                self.separate_artists()
  1107→                self.create_buffer()
  1108→
  1109→        if len(self.buffer) > 0:
  1110→            return self.buffer.pop()
  1111→        else:
  1112→            return None
