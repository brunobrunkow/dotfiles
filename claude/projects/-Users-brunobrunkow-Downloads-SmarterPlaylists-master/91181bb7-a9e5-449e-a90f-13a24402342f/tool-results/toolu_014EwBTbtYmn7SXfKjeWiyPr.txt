   104→class TrackFilter(object):
   105→    '''
   106→        produces tracks on the true source that are not on the false source
   107→    '''
   108→    def __init__(self, true_source, false_source, invert=False, by_name=False):
   109→        prep = ' that are also in ' if invert else ' that are not in '
   110→        self.name = 'tracks in ' + true_source.name  + prep + \
   111→            false_source.name
   112→        self.true_source = true_source
   113→        self.false_source = false_source
   114→        self.bad_tracks = set()
   115→        self.invert = invert
   116→        self.debug = False
   117→        self.by_name = by_name
   118→
   119→    def next_track(self):
   120→        while True:
   121→            bad_track = self.false_source.next_track()
   122→            self.bad_tracks.add(bad_track)
   123→            if bad_track:
   124→                if self.by_name:
   125→                    bad_track_name = pbl.tlib.get_tn(bad_track).lower()
   126→                    self.bad_tracks.add(bad_track_name)
   127→            else:
   128→                break
   129→
   130→        while True:
   131→            track = self.true_source.next_track()
   132→            if track:
   133→                if self.by_name:
   134→                    track_name = pbl.tlib.get_tn(track).lower()
   135→                else:
   136→                    track_name = track
   137→
   138→                if self.invert and ((track in self.bad_tracks) or (track_name in self.bad_tracks)):
   139→                    return track
   140→                elif (not self.invert) and ((track not in self.bad_tracks) and (track_name not in self.bad_tracks)):
   141→                    return track
   142→                else:
   143→                    if self.debug:
   144→                        print('filtered out', pbl.tlib.get_tn(track))
   145→            else:
   146→                break
   147→        return None

   187→class TextFilter(object):
   188→    '''
   189→        produces tracks from the stream based on track title match
   190→    '''
   191→    def __init__(self, source, text, ignore_case=False, invert=False):
   192→        prep = ' that do not match ' if invert else ' that match '
   193→        self.name = 'tracks in ' + source.name  + prep + '"' + text + '"'
   194→        self.source = source
   195→        self.invert = invert
   196→        flags = re.UNICODE
   197→        if ignore_case:
   198→            flags |= re.IGNORECASE
   199→        self.regex = re.compile(text, flags)
   200→
   201→    def next_track(self):
   202→        while True:
   203→            track = self.source.next_track()
   204→            if track:
   205→                tinfo = pbl.tlib.get_track(track)
   206→                title = tinfo['title']
   207→                does_match = self.regex.search(title) != None
   208→                if does_match == self.invert:
   209→                    continue
   210→                else:
   211→                    return track
   212→            else:
   213→                break
   214→        return None

   300→class AllButTheFirst(object):
   301→    '''
   302→        Returns all but the first tracks from a stream
   303→
   304→        :param source: the source of tracks
   305→        :param sample_size: the number of tracks to skip
   306→    '''
   307→    def __init__(self, source, sample_size=10):
   308→        self.name = 'all but the first ' + str(sample_size) + ' of ' + source.name
   309→        self.source = source
   310→        self.sample_size = sample_size
   311→        self.buffer = []
   312→        self.filling = True
   313→
   314→    def next_track(self):
   315→        while self.filling:
   316→            track = self.source.next_track()
   317→            if track:
   318→                self.buffer.append(track)
   319→            else:
   320→                self.filling = False
   321→                self.buffer = self.buffer[self.sample_size:]
   322→
   323→        if len(self.buffer) > 0:
   324→            return self.buffer.pop(0)
   325→        else:
   326→            return None

   401→            self.buffer.append(track)
   402→        elif not self.saved:
   403→            self._save_playlist()

   423→            if not user:
   424→                raise pbl.PBLException(self, "bad uri")
   425→
   426→            if not pid:

   438→        pid = get_pid_from_playlist_uri(uri)

   972→    '''
   973→        A PBL Filter that mixes two input streams based upon a small
   974→        set of rules
   975→    '''
   976→
   977→    def __init__(self, true_source, false_source,
   978→        ntracks=1, nskips=1, initial_skip = 1, fail_fast = True):
   979→        '''
   980→            params:
   981→                * true_source
   982→                * false_source
   983→        '''
   984→        self.name = 'mixing ' + true_source.name  + ' with ' + \
   985→            false_source.name
   986→        self.true_source = true_source
   987→        self.false_source = false_source
   988→        self.initial_offset = initial_skip
   989→        self.ntracks = ntracks # false tracks in a row
   990→        self.nskips = nskips   # true tracks in a row
   991→        self.fail_fast = fail_fast
   992→        self.which = 0

   994→    def next_track(self):
   995→
   996→        if self.which < self.initial_offset:
   997→            true_count = self.initial_offset
   998→        else:
   999→            true_count = self.nskips
  1000→
  1001→        false_count = self.ntracks
  1002→        total_tracks_in_cycle = true_count + false_count
  1003→        which_in_cycle = self.which % total_tracks_in_cycle
  1004→        use_first = which_in_cycle < true_count
  1005→
  1006→        if use_first:
  1007→            next_source = self.true_source
  1008→            other_source = self.false_source
  1009→        else:
  1010→            next_source = self.false_source
  1011→            other_source = self.true_source
  1012→
  1013→        track = next_source.next_track()
  1014→        if not track:
  1015→            if not self.fail_fast:
  1016→                track = other_source.next_track()
  1017→        self.which += 1
  1018→        return track

  1020→class SeparateArtists(object):
  1021→    ''' A PBL filter that reorders the input tracks to maximize
  1022→        the separation between artists
  1023→    '''
  1024→
  1025→    def __init__(self, source):
  1026→        self.name = 'SeparateArtists'
  1027→        self.source = source
  1028→        self.tracks = []
  1029→        self.filling = True
  1030→
  1031→    def score_list(self):
  1032→        min_delta_allowed = 2
  1033→        score = 0
  1034→        indexes = set()
  1035→
  1036→        for i in range(len(self.tracks) - 1):
  1037→            aname = self.tracks[i]['artist']
  1038→            for j in range(i + 1, len(self.tracks)):
  1039→
  1040→                if j - i >= min_delta_allowed:
  1041→                    break
  1042→
  1043→                bname = self.tracks[j]['artist']
  1044→                if aname == bname:
  1045→                    delta = j - i
  1046→                    indexes.add(i)
  1047→                    indexes.add(j)
  1048→                    score += min_delta_allowed - delta
  1049→
  1050→        lindex = list(indexes)
  1051→        lindex.sort()
  1052→        return score, lindex
  1053→
  1054→    def random_index(self):
  1055→        return random.randint(0, len(self.tracks) - 1)
  1056→
  1057→    def swap(self, a, b):
  1058→        tmp = self.tracks[a]
  1059→        self.tracks[a] = self.tracks[b]
  1060→        self.tracks[b] = tmp
  1061→
  1062→    def separate_artists(self):
  1063→        max_tries = 1000
  1064→        cur_score = 0
  1065→        max_no_swaps = 100
  1066→
  1067→        cur_score, indexes = self.score_list()
  1068→        no_swap = 0
  1069→
  1070→        for i in range(max_tries):
  1071→
  1072→            if cur_score == 0 or len(indexes) == 0:
  1073→                break
  1074→
  1075→            swap_1 = random.choice(indexes)
  1076→            swap_2 = self.random_index()
  1077→            self.swap(swap_1, swap_2)
  1078→            new_score, new_indexes = self.score_list()
  1079→
  1080→            if new_score >= cur_score:
  1081→                self.swap(swap_2, swap_1)
  1082→                no_swap += 1
  1083→                if no_swap > max_no_swaps:
  1084→                    break
  1085→            else:
  1086→                no_swap = 0
  1087→                cur_score = new_score
  1088→                indexes = new_indexes
  1089→
  1090→            if cur_score == 0:
  1091→                break
  1092→
  1093→        return cur_score, len(indexes)
  1094→
  1095→    def create_buffer(self):
  1096→        self.buffer = []
  1097→        for ti in self.tracks:
  1098→            self.buffer.append(ti['id'])

  1100→    def next_track(self):
  1101→        while self.filling:
  1102→            track = self.source.next_track()
  1103→            if track:
  1104→                tinfo = pbl.tlib.get_track(track)
  1105→                self.tracks.append(tinfo)
  1106→            else:
  1107→                self.filling = False
  1108→                self.separate_artists()
  1109→                self.create_buffer()
  1110→
  1111→        if len(self.buffer) > 0:
  1112→            return self.buffer.pop()
  1113→        else:
  1114→            return None
